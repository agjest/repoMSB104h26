Predicting log models

house_sp <- spData::house |> 
  as.data.frame() |> 
  select(price, yrbuilt, stories, TLA, beds, baths, lotsize, syear) |> 
  filter(syear == 1997)
m_lev_lev <- paste0(
  'price ~ yrbuilt + TLA + ',
  'beds + baths + lotsize'
)
m_log_lev <- paste0(
  'log(price) ~ yrbuilt + TLA + ',
  'beds + baths + lotsize'
)
house_sp |> 
  ggplot(
    mapping = aes(x = price)
  ) +
  geom_histogram(binwidth = 10000)
house_sp |> 
  ggplot(
    mapping = aes(x = log(price))
  ) +
  geom_histogram(binwidth = 0.1)
reg_lev_lev <- house_sp %>%
  lm(
    formula = m_lev_lev,
    data = .
  )
reg_lev_lev |> 
  flextable::as_flextable() |> 
  flextable::line_spacing(space = 0.3, part = "all")
