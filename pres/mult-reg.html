# Data for constructed example
df <- data.frame(W = rep(c(FALSE, TRUE), each = 100))  |> 
  mutate(
    X = .5 + 2 * W + rnorm(200),
    Y = -.5 * X + 4 * W + 1 + rnorm(200)
    )
  
df2 <- df |> 
  group_by(W) |> 
  mutate(
    mean_X = mean(X), 
    mean_Y = mean(Y)
    )  |> 
  ungroup()
df2 |> 
  ggplot(
    mapping = aes(y = Y, x = X)
    ) +
  geom_point(size = 3) +
  geom_smooth(
    formula = 'y ~ x',
    method = lm, 
    se = FALSE, 
    color = "blue"
    )
df2 |> 
  mutate(W = fct(as.character(W))) |> 
  ggplot(
    mapping = aes(y = Y, x = X, color = W)
    ) +
  geom_point(size = 3) +
  geom_smooth(
    formula = 'y ~ x',
    method = lm, 
    se = FALSE, 
    color = "blue"
    )
df2 |> 
  mutate(
    W = fct(as.character(W))
    ) |> 
  ggplot(
    mapping = aes(
      y = Y, 
      x = X, 
      color = W
      )
    ) +
  geom_point(size = 3) +
  geom_smooth(
    formula = 'y ~ x',
    method = lm, 
    se = FALSE, 
    color = "blue"
    ) +
  facet_wrap(
    '~ W'
  )
mtcars_inter <- mtcars |> 
  select(mpg, wt, cyl) |> 
  mutate(
    dum_6cyl = ifelse(cyl == 6, 1, 0),
    dum_8cyl = ifelse(cyl == 8, 1, 0),
    wt_6cyl = wt * dum_6cyl,
    wt_8cyl = wt * dum_8cyl
  )
lm_inter <- lm(
  formula = as.formula(
    paste("mpg",
          paste("wt", "dum_6cyl",
                "dum_8cyl", "wt_6cyl", 
                "wt_8cyl",
                sep = " + "
                ),
          sep = " ~ "
          )
    ),
  data = mtcars_inter
  )

summary(lm_inter)
